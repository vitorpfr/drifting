# Drifting Web — Visualization Vertical Slice Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Prove the full Drifting web tech stack end-to-end: catalog loads from a precomputed JSON, a random station plays audio, and the correct visualization mode (audio-reactive or ambient) activates in a Three.js WebGL canvas.

**Architecture:** React + Vite wraps a full-screen Three.js canvas. Two renderer classes (`AmbientRenderer`, `AudioReactiveRenderer`) share a single `WebGLRenderer`. `VisualizationManager` picks the active renderer from the station's `corsFriendly` flag and falls back to ambient if the `AnalyserNode` stays silent for 2 seconds. A standalone Node script (`buildCatalog.ts`) precomputes the catalog by HEAD-checking ~2000 filtered radio-browser.info stations for CORS.

**Tech Stack:** React 18, Vite 5, TypeScript, Three.js r168, Vitest, tsx (for running scripts)

> **Vertical slice scope:** Mode switching in this plan is an instant cut (no fade). The 300 ms cross-fade transition described in the spec is deferred to the next iteration. Everything else in the spec is implemented.

```typescript
import { writeFileSync, mkdirSync } from 'fs'
import { existsSync } from 'fs'
import path from 'path'

const API_BASE = 'https://de1.api.radio-browser.info/json/stations/search'
const CONCURRENCY = 20
const TIMEOUT_MS = 6000

const MUSIC_TAGS = [
  'music', 'pop', 'rock', 'jazz', 'classical', 'electronic',
  'hip-hop', 'country', 'rnb', 'soul', 'blues', 'folk', 'indie',
]

interface RawStation {
  stationuuid: string
  name: string
  url: string
  url_resolved: string
  tags: string
  country: string
  bitrate: number
  lastcheckok: number
}

export interface CatalogStation {
  stationuuid: string
  name: string
  url: string
  tags: string
  country: string
  bitrate: number
  corsFriendly: boolean
}

export function isMusicStation(tags: string): boolean {
  const t = tags.toLowerCase()
  return MUSIC_TAGS.some(tag => t.includes(tag))
}

async function fetchStations(limit: number): Promise<RawStation[]> {
  const params = new URLSearchParams({
    limit: String(limit),
    bitrateMin: '128',
    lastcheckok: '1',
    order: 'votes',
    reverse: 'true',
  })
  const res = await fetch(`${API_BASE}?${params}`, {
    headers: { 'User-Agent': 'drifting-catalog-builder/1.0' },
  })
  if (!res.ok) throw new Error(`API error: ${res.status}`)
  const all: RawStation[] = await res.json()
  return all.filter(s => isMusicStation(s.tags))
}

async function checkCors(url: string): Promise<boolean> {
  const controller = new AbortController()
  const timer = setTimeout(() => controller.abort(), TIMEOUT_MS)
  try {
    const res = await fetch(url, { method: 'HEAD', signal: controller.signal })
    clearTimeout(timer)
    const acao = res.headers.get('access-control-allow-origin')
    return acao !== null
  } catch {
    clearTimeout(timer)
    return false
  }
}

async function runWithConcurrency<T, R>(
  items: T[],
  fn: (item: T, index: number) => Promise<R>,
  concurrency: number,
): Promise<R[]> {
  const results: R[] = new Array(items.length)
  for (let i = 0; i < items.length; i += concurrency) {
    const batch = items.slice(i, i + concurrency)
    const batchResults = await Promise.all(batch.map((item, j) => fn(item, i + j)))
    batchResults.forEach((r, j) => { results[i + j] = r })
    process.stdout.write(`\rCORS check: ${Math.min(i + concurrency, items.length)}/${items.length}`)
  }
  process.stdout.write('\n')
  return results
}

async function main() {
  const limitArg = process.argv.find(a => a.startsWith('--limit='))
  const limit = limitArg ? parseInt(limitArg.split('=')[1], 10) : 10000

  console.log(`Fetching up to ${limit} stations from radio-browser.info…`)
  const stations = await fetchStations(limit)
  console.log(`Found ${stations.length} music stations with ≥128 kbps`)

  console.log('Checking CORS headers…')
  const corsResults = await runWithConcurrency(
    stations,
    station => checkCors(station.url_resolved || station.url),
    CONCURRENCY,
  )

  const catalog: CatalogStation[] = stations.map((s, i) => ({
    stationuuid: s.stationuuid,
    name: s.name,
    url: s.url_resolved || s.url,
    tags: s.tags,
    country: s.country,
    bitrate: s.bitrate,
    corsFriendly: corsResults[i],
  }))

  const corsCount = corsResults.filter(Boolean).length
  console.log(`CORS-friendly: ${corsCount}/${stations.length} (${Math.round((corsCount / stations.length) * 100)}%)`)

  const outDir = path.resolve(process.cwd(), 'public')
  if (!existsSync(outDir)) mkdirSync(outDir, { recursive: true })
  writeFileSync(path.join(outDir, 'catalog.json'), JSON.stringify(catalog))
  console.log(`Written to public/catalog.json`)
}

main().catch(err => { console.error(err); process.exit(1) })
```

```typescript
const GENRE_COLORS: Record<string, Color> = {
  jazz:       { r: 0.90, g: 0.60, b: 0.10 },
```

```typescript
import type { FrequencyBuckets } from './types'

const FFT_SIZE = 2048

export class AudioPlayer {
  private audio: HTMLAudioElement
  private context: AudioContext | null = null
  private analyser: AnalyserNode | null = null
  private dataArray: Uint8Array | null = null

  constructor() {
    this.audio = new Audio()
  }

  async play(url: string, withCors: boolean): Promise<void> {
    this.audio.pause()
    if (withCors) {
      this.audio.crossOrigin = 'anonymous'
    } else {
      this.audio.removeAttribute('crossorigin')
    }
    this.audio.src = url

    if (withCors && !this.context) {
      this.context = new AudioContext()
      this.analyser = this.context.createAnalyser()
      this.analyser.fftSize = FFT_SIZE
      this.dataArray = new Uint8Array(this.analyser.frequencyBinCount)
      const source = this.context.createMediaElementSource(this.audio)
      source.connect(this.analyser)
      this.analyser.connect(this.context.destination)
    }

    if (this.context?.state === 'suspended') {
      await this.context.resume()
    }

    await this.audio.play()
  }

  stop(): void {
    this.audio.pause()
    this.audio.src = ''
  }

  getFrequencyBuckets(): FrequencyBuckets {
    if (!this.analyser || !this.dataArray) return { bass: 0, mid: 0, treble: 0 }

    this.analyser.getByteFrequencyData(this.dataArray)

    const sampleRate = this.context?.sampleRate ?? 44100
    const binHz = sampleRate / FFT_SIZE
    const bassBins = Math.floor(250 / binHz)
    const midBins = Math.floor(4000 / binHz)
    const total = this.dataArray.length

    return {
      bass:   avg(this.dataArray, 0, bassBins),
      mid:    avg(this.dataArray, bassBins, midBins),
      treble: avg(this.dataArray, midBins, total),
    }
  }

  isSilent(): boolean {
    if (!this.analyser || !this.dataArray) return true
    this.analyser.getByteFrequencyData(this.dataArray)
    return this.dataArray.every(v => v === 0)
  }
}

function avg(data: Uint8Array, start: number, end: number): number {
  if (end <= start) return 0
  let sum = 0
  for (let i = start; i < end; i++) sum += data[i]
  return sum / (end - start) / 255
}
```

```typescript
import * as THREE from 'three'
import { getGenreColor } from './genreColors'

const VERT = /* glsl */`
varying vec2 vUv;
void main() {
  vUv = uv;
  gl_Position = vec4(position, 1.0);
}
`

const FRAG = /* glsl */`
uniform float u_time;
uniform vec3 u_color;
uniform float u_opacity;
varying vec2 vUv;

float hash(vec2 p) {
  return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453);
}

float noise(vec2 p) {
  vec2 i = floor(p);
  vec2 f = fract(p);
  vec2 u = f * f * (3.0 - 2.0 * f);
  return mix(
    mix(hash(i),            hash(i + vec2(1.0, 0.0)), u.x),
    mix(hash(i + vec2(0.0, 1.0)), hash(i + vec2(1.0, 1.0)), u.x),
    u.y
  );
}

float fbm(vec2 p) {
  float v = 0.0;
  float a = 0.5;
  for (int i = 0; i < 5; i++) {
    v += a * noise(p);
    p = p * 2.0 + vec2(1.7, 9.2);
    a *= 0.5;
  }
  return v;
}

void main() {
  float t = u_time * 0.08;
  vec2 q = vec2(fbm(vUv + t), fbm(vUv + vec2(1.0)));
  vec2 r = vec2(
    fbm(vUv + q + vec2(1.7, 9.2) + 0.15 * t),
    fbm(vUv + q + vec2(8.3, 2.8) + 0.126 * t)
  );
  float f = fbm(vUv + r);
  float brightness = pow(clamp(f, 0.0, 1.0), 1.5) * 1.4;
  gl_FragColor = vec4(u_color * brightness, u_opacity);
}
`

export class AmbientRenderer {
  private scene: THREE.Scene
  private camera: THREE.OrthographicCamera
  private material: THREE.ShaderMaterial
  private targetColor = new THREE.Color(0.1, 0.6, 0.6)
  private currentColor = new THREE.Color(0.1, 0.6, 0.6)
  private colorTransitionStart = -Infinity
  private readonly colorTransitionDuration = 2000

  constructor() {
    this.scene = new THREE.Scene()
    this.camera = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1)

    this.material = new THREE.ShaderMaterial({
      vertexShader: VERT,
      fragmentShader: FRAG,
      transparent: true,
      uniforms: {
        u_time:    { value: 0 },
        u_color:   { value: new THREE.Vector3(0.1, 0.6, 0.6) },
        u_opacity: { value: 1 },
      },
    })

    this.scene.add(new THREE.Mesh(new THREE.PlaneGeometry(2, 2), this.material))
  }

  setMetadata(tags: string, _country: string): void {
    const c = getGenreColor(tags)
    this.targetColor.setRGB(c.r, c.g, c.b)
    this.colorTransitionStart = performance.now()
  }

  setOpacity(opacity: number): void {
    this.material.uniforms.u_opacity.value = opacity
  }

  animate(renderer: THREE.WebGLRenderer, elapsedMs: number): void {
    this.material.uniforms.u_time.value = elapsedMs / 1000

    const progress = Math.min(
      (performance.now() - this.colorTransitionStart) / this.colorTransitionDuration,
      1,
    )
    this.currentColor.lerp(this.targetColor, progress)
    this.material.uniforms.u_color.value.set(
      this.currentColor.r,
      this.currentColor.g,
      this.currentColor.b,
    )

    renderer.render(this.scene, this.camera)
  }

  dispose(): void {
    this.material.dispose()
  }
}
```

```typescript
import * as THREE from 'three'
import type { Color, FrequencyBuckets } from '../types'

const PARTICLE_COUNT = 8000

const VERT = /* glsl */`
attribute float a_seed;
uniform float u_bass;
uniform float u_mid;
uniform float u_treble;
uniform float u_time;
uniform float u_opacity;

varying float vAlpha;

void main() {
  float angle = a_seed * 6.2831853 + u_time * (0.3 + u_mid * 1.2);
  float band   = mod(a_seed * 7.0, 1.0);
  float radius = (0.05 + band * 0.85) * (0.4 + u_bass * 0.6);

  vec3 pos = vec3(cos(angle) * radius, sin(angle) * radius, 0.0);
  gl_Position  = vec4(pos, 1.0);
  gl_PointSize = 1.5 + u_treble * 3.0;
  vAlpha = (0.4 + u_treble * 0.6) * u_opacity;
}
`

const FRAG = /* glsl */`
uniform vec3 u_color;
varying float vAlpha;

void main() {
  float d = length(gl_PointCoord - 0.5);
  if (d > 0.5) discard;
  float alpha = (1.0 - d * 2.0) * vAlpha;
  gl_FragColor = vec4(u_color, alpha);
}
`

export class AudioReactiveRenderer {
  private scene: THREE.Scene
  private camera: THREE.OrthographicCamera
  private material: THREE.ShaderMaterial
  private elapsedSec = 0

  constructor() {
    this.scene = new THREE.Scene()
    this.camera = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1)

    const geometry = new THREE.BufferGeometry()
    const seeds = new Float32Array(PARTICLE_COUNT)
    const positions = new Float32Array(PARTICLE_COUNT * 3)
    for (let i = 0; i < PARTICLE_COUNT; i++) {
      seeds[i] = i / PARTICLE_COUNT
    }
    geometry.setAttribute('position',  new THREE.BufferAttribute(positions, 3))
    geometry.setAttribute('a_seed',    new THREE.BufferAttribute(seeds, 1))

    this.material = new THREE.ShaderMaterial({
      vertexShader:   VERT,
      fragmentShader: FRAG,
      transparent:    true,
      depthWrite:     false,
      uniforms: {
        u_bass:    { value: 0 },
        u_mid:     { value: 0 },
        u_treble:  { value: 0 },
        u_time:    { value: 0 },
        u_color:   { value: new THREE.Vector3(0.1, 0.6, 0.6) },
        u_opacity: { value: 1 },
      },
    })

    this.scene.add(new THREE.Points(geometry, this.material))
  }

  setFrequencyBuckets(buckets: FrequencyBuckets): void {
    this.material.uniforms.u_bass.value   = buckets.bass
    this.material.uniforms.u_mid.value    = buckets.mid
    this.material.uniforms.u_treble.value = buckets.treble
  }

  setGenreColor(color: Color): void {
    this.material.uniforms.u_color.value.set(color.r, color.g, color.b)
  }

  setOpacity(opacity: number): void {
    this.material.uniforms.u_opacity.value = opacity
  }

  animate(renderer: THREE.WebGLRenderer, deltaMs: number): void {
    this.elapsedSec += deltaMs / 1000
    this.material.uniforms.u_time.value = this.elapsedSec
    renderer.render(this.scene, this.camera)
  }

  dispose(): void {
    this.material.dispose()
  }
}
```

```typescript
import * as THREE from 'three'
import type { Station } from '../types'
import type { AudioPlayer } from '../audioPlayer'
import { AmbientRenderer } from './AmbientRenderer'
import { AudioReactiveRenderer } from './AudioReactiveRenderer'
import { getGenreColor } from './genreColors'

export type RendererMode = 'ambient' | 'audio-reactive'

const SILENCE_GRACE_MS = 2000

export class VisualizationManager {
  private webglRenderer: THREE.WebGLRenderer
  private ambientRenderer: AmbientRenderer
  private audioReactiveRenderer: AudioReactiveRenderer
  private currentPlayer: AudioPlayer | null = null
  private silenceTimer: ReturnType<typeof setTimeout> | null = null
  private clock = new THREE.Clock()
  private rafHandle = 0
  activeMode: RendererMode = 'ambient'

  constructor(canvas: HTMLCanvasElement) {
    this.webglRenderer = new THREE.WebGLRenderer({ canvas, antialias: true })
    this.webglRenderer.setPixelRatio(window.devicePixelRatio)
    this.webglRenderer.setSize(window.innerWidth, window.innerHeight)
    this.ambientRenderer = new AmbientRenderer()
    this.audioReactiveRenderer = new AudioReactiveRenderer()
  }

  loadStation(station: Station, player: AudioPlayer): void {
    this.currentPlayer = player
    if (this.silenceTimer) {
      clearTimeout(this.silenceTimer)
      this.silenceTimer = null
    }

    if (station.corsFriendly) {
      this.activeMode = 'audio-reactive'
      this.audioReactiveRenderer.setGenreColor(getGenreColor(station.tags))
      this.silenceTimer = setTimeout(() => {
        if (this.currentPlayer?.isSilent()) {
          this.activeMode = 'ambient'
          this.ambientRenderer.setMetadata(station.tags, station.country)
        }
      }, SILENCE_GRACE_MS)
    } else {
      this.activeMode = 'ambient'
      this.ambientRenderer.setMetadata(station.tags, station.country)
    }
  }

  start(): void {
    this.clock.start()
    const loop = () => {
      this.rafHandle = requestAnimationFrame(loop)
      this.tick()
    }
    loop()
  }

  stop(): void {
    cancelAnimationFrame(this.rafHandle)
  }

  private tick(): void {
    const elapsedMs  = this.clock.getElapsedTime() * 1000
    const deltaMs    = this.clock.getDelta() * 1000

    this.webglRenderer.autoClear = true

    if (this.activeMode === 'audio-reactive') {
      if (this.currentPlayer) {
        this.audioReactiveRenderer.setFrequencyBuckets(
          this.currentPlayer.getFrequencyBuckets(),
        )
      }
      this.audioReactiveRenderer.setOpacity(1)
      this.audioReactiveRenderer.animate(this.webglRenderer, deltaMs)
    } else {
      this.ambientRenderer.setOpacity(1)
      this.ambientRenderer.animate(this.webglRenderer, elapsedMs)
    }
  }

  dispose(): void {
    this.stop()
    if (this.silenceTimer) clearTimeout(this.silenceTimer)
    this.ambientRenderer.dispose()
    this.audioReactiveRenderer.dispose()
    this.webglRenderer.dispose()
  }
}
```

```tsx
import { useEffect, useRef, useState } from 'react'
import { loadCatalog, getRandomStation } from './catalog'
import { AudioPlayer } from './audioPlayer'
import { VisualizationManager } from './visualization/VisualizationManager'
import type { Station } from './types'

export default function App() {
  const canvasRef  = useRef<HTMLCanvasElement>(null)
  const vizRef     = useRef<VisualizationManager | null>(null)
  const playerRef  = useRef<AudioPlayer | null>(null)
  const [station, setStation]   = useState<Station | null>(null)
  const [started, setStarted]   = useState(false)
  const [loading, setLoading]   = useState(false)
  const [error, setError]       = useState<string | null>(null)

  // Set up VisualizationManager on mount (before any interaction)
  useEffect(() => {
    if (!canvasRef.current) return
    const viz = new VisualizationManager(canvasRef.current)
    vizRef.current = viz
    viz.start()
    return () => viz.dispose()
  }, [])

  const handleStart = async () => {
    setLoading(true)
    setError(null)
    try {
      const stations = await loadCatalog()
      const s = getRandomStation(stations)
      setStation(s)

      const player = new AudioPlayer()
      playerRef.current = player
      await player.play(s.url, s.corsFriendly)

      vizRef.current?.loadStation(s, player)
      setStarted(true)
    } catch (e) {
      setError(String(e))
    } finally {
      setLoading(false)
    }
  }

  return (
    <>
      <canvas
        ref={canvasRef}
        style={{ position: 'fixed', inset: 0, width: '100%', height: '100%' }}
      />

      {!started && (
        <div style={{
          position: 'fixed', inset: 0,
          display: 'flex', alignItems: 'center', justifyContent: 'center',
          background: 'rgba(0,0,0,0.7)',
        }}>
          <button
            onClick={handleStart}
            disabled={loading}
            style={{
              padding: '16px 40px',
              fontSize: 18,
              background: 'white',
              color: 'black',
              border: 'none',
              borderRadius: 8,
              cursor: loading ? 'wait' : 'pointer',
            }}
          >
            {loading ? 'Loading…' : 'Start Drifting'}
          </button>
          {error && (
            <p style={{ color: 'red', marginTop: 16, position: 'absolute', bottom: 40 }}>
              {error}
            </p>
          )}
        </div>
      )}

      {started && station && (
        <div style={{
          position: 'fixed', bottom: 32,
          left: 0, right: 0, textAlign: 'center',
          color: 'rgba(255,255,255,0.8)',
          fontFamily: 'system-ui, sans-serif',
          fontSize: 16,
          pointerEvents: 'none',
          textShadow: '0 1px 4px rgba(0,0,0,0.8)',
        }}>
          {station.name} · {station.country} · {station.corsFriendly ? '🎵 live analysis' : '🌊 ambient'}
        </div>
      )}
    </>
  )
}
```
